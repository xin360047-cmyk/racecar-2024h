mod_gimbal.c — 舵机云台控制
/* 舵机PWM：50Hz周期，高电平 0.5ms~2.5ms = 角度0°~180° */ #define SERVO_MIN 500 // 0.5ms = 500μs #define SERVO_MAX 2500 // 2.5ms = 2500μs void mod_gimbal_set(uint8_t h_deg, uint8_t v_deg) { uint16_t pulse_h = SERVO_MIN + h_deg * (SERVO_MAX - SERVO_MIN) / 180; uint16_t pulse_v = SERVO_MIN + v_deg * (SERVO_MAX - SERVO_MIN) / 180; bsp_pwm_set_pulse(CH_SERVO_H, pulse_h); bsp_pwm_set_pulse(CH_SERVO_V, pulse_v); } /* 效果：h_deg=90 → 水平舵机转到中间位置 * 说明：60°舵机和180°舵机的高电平范围不同，上面以180°舵机为例 */
main.c — 双线主循环
#include "bsp/..." #include "module/..." #include "app/task_car.h" #include "app/task_gimbal.h" void main(void) { bsp_all_init(); // 所有外设初始化 mod_gimbal_init(); // 舵机归零到90° while (BTN_START()); // 等按键 while (1) { task_car(); // 小车循迹控制 task_gimbal(); // 云台瞄准控制 delay_ms(10); // 每10ms跑一次 } }
task_car.c — 小车控制（开环为主）
void task_car(void) { switch (g_car_state) { case CAR_STRAIGHT: mod_motor_set(BASE_SPEED, BASE_SPEED); // 直走 if (mod_encoder_get_distance() >= SEG_LEN) { mod_encoder_reset(); g_car_state = CAR_TURN; } break; case CAR_TURN: mod_motor_set(TURN_SPEED, 0); // 左轮转右轮停=右转 if (mod_encoder_get_distance() >= TURN_DIST) { mod_encoder_reset(); g_car_state = CAR_STRAIGHT; g_segment++; if (g_segment == 4) { // 回到A→换圈 g_segment = 0; g_lap++; if (g_lap >= g_target_laps) g_car_state = CAR_STOP; } } break; case CAR_STOP: mod_motor_stop(); break; } }
task_gimbal.c — 云台控制
void task_gimbal(void) { switch (g_gimbal_state) { case GIMBAL_IDLE: break; // 等待瞄准开关接通 case GIMBAL_AIM: // 根据小车当前位置(g_segment, distance)计算目标角度 coord_get_angles(g_segment, g_distance, &h, &v); mod_gimbal_set(h, v); break; case GIMBAL_CIRCLE: // 画圆：根据小车行程比例计算相位 → 水平角度和俯仰角度做正弦/余弦 float phase = coord_get_circle_phase(g_lap, g_segment, g_distance); mod_gimbal_set( center_h + CIRCLE_RADIUS_DEG * sin(phase), center_v + CIRCLE_RADIUS_DEG * cos(phase) ); break; } }
